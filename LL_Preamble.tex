%% Type
%% ---------------------------------------------------------------------------
%% Two cuts of the same French-Renaissance letterform, each doing one job.
%%
%% EB Garamond is the text cut: it carries the ink and the wider fitting a
%% document reads at 11pt.  Cormorant Garamond is the display cut of the same
%% design -- finer strokes, higher contrast, drawn for large sizes -- and it
%% sets the name and the section headings, where those qualities are the point.
%% (Cormorant alone set the whole CV before, and went thin and grey in the body
%% text at reading size.)
%%
%% ebgaramond-maths puts the math in the same family, so the $\sqrt{s}$ and the
%% $\tau^+\tau^-$ in the publication titles stop falling back to Computer
%% Modern, which is a visibly different letterform mid-title.

%% Oldstyle (lowercase) figures throughout: they sit at x-height with ascenders
%% and descenders, so a year reads as a word inside a line of prose instead of
%% as a row of capitals.  It is the book setting, and this document is mostly
%% prose with dates in it.
\usepackage[oldstyle]{ebgaramond}  % text face, oldstyle figures
\usepackage{ebgaramond-maths}     % math in the same family

% \displayface -- the Cormorant Garamond cut, for the name and headings.
% CormorantGaramond-LF is the lining-figure family its package installs; taking
% it with \usefont keeps Cormorant out of \rmdefault, so it never leaks into
% body text.
\newcommand{\displayface}{\usefont{T1}{CormorantGaramond-OsF}{m}{n}}

\usepackage{microtype}            % subtler justification, fewer overfull lines

% EB Garamond sets fi/fl as single ligature glyphs, which the PDF carries as one
% character.  Without a ToUnicode map "configuration" copies out as
% "con<FB01>guration"; with the stock map it copies as the U+FB01 ligature
% character, which is correct Unicode but still does not match a search for
% "configuration" -- and a CV gets read by keyword searches as often as by
% people.  So map each ligature to the letters it stands for: it renders as a
% ligature and copies as plain text.
% EB Garamond's cuts do not agree on what to call these glyphs -- the roman
% encoding uses f_i, the italic uniFB01 -- so all three spellings are mapped.
\input{glyphtounicode}
\pdfgentounicode=1
\pdfglyphtounicode{ff}      {0066 0066}
\pdfglyphtounicode{fi}      {0066 0069}
\pdfglyphtounicode{fl}      {0066 006C}
\pdfglyphtounicode{ffi}     {0066 0066 0069}
\pdfglyphtounicode{ffl}     {0066 0066 006C}
\pdfglyphtounicode{f_f}     {0066 0066}
\pdfglyphtounicode{f_i}     {0066 0069}
\pdfglyphtounicode{f_l}     {0066 006C}
\pdfglyphtounicode{f_f_i}   {0066 0066 0069}
\pdfglyphtounicode{f_f_l}   {0066 0066 006C}
\pdfglyphtounicode{uniFB00} {0066 0066}
\pdfglyphtounicode{uniFB01} {0066 0069}
\pdfglyphtounicode{uniFB02} {0066 006C}
\pdfglyphtounicode{uniFB03} {0066 0066 0069}
\pdfglyphtounicode{uniFB04} {0066 0066 006C}

%% ===========================================================================
%% Internal and external editions
%% ===========================================================================
%% The internal edition carries material that is not for general circulation:
%% the current student and postdoc roster, thesis committee membership,
%% individual refereeing assignments, and student travel awards.  The external
%% edition is the default and needs no flag; the internal one is selected by
%% defining \CVinternal before the document is read:
%%
%%     pdflatex -jobname=LL_CV_internal "\def\CVinternal{}\input{LL_CV.tex}"
%%
%% which is what build.sh and .github/workflows/build-cv.yml do.

\newif\ifCVinternal
\ifdefined\CVinternal \CVinternaltrue \else \CVinternalfalse \fi

% \internalonly{<content>} -- included only in the internal edition.
% \externalonly{<content>} -- included only in the external edition.
%
% Both end by removing a trailing word space (when there is one to remove).
% The newlines above and below a block each become a space token, and the
% newline before its closing brace becomes one more inside it; TeX discards
% only the final glue of a paragraph, so without the cleanup one survivor was
% left after the date box and pushed the date a word space left of its column
% -- so for UT-Battelle above a dropped block in the external edition, and for
% the last entry inside a kept block in the internal one.
\newcommand{\internalonly}[1]{\ifCVinternal #1\fi\ifhmode\unskip\fi}
\newcommand{\externalonly}[1]{\ifCVinternal\else #1\fi\ifhmode\unskip\fi}

% The internal edition says so in its PDF title, so the two files stay

\fi

%% ===========================================================================
%% Lists
%% ===========================================================================
%% The CV is built almost entirely from bullet-free lists that differ only in
%% how tightly they are set.  itemize itself is redefined, rather than
%% configured through enumitem, so that a plain \begin{itemize} anywhere picks
%% up the CV style.
%%
%%   itemize               default spacing between entries
%%   itemizetight          no extra space between entries
%%   itemizetighter        entries set as close as the leading allows
%%   itemizetightrightpad  tight, and inset from the right margin, for the
%%                         commentary set underneath an entry
%%
%% \LLlist{<left>}{<right>}{<itemsep>}{<parskip>}{<parsep>}

\newcommand{\LLlist}[5]{%
  \list{}{%
    \setlength{\leftmargin}{#1}%
    \setlength{\rightmargin}{#2}%
    \setlength{\itemsep}{#3}%
    \setlength{\parskip}{#4}%
    \setlength{\parsep}{#5}%
  }%
}

\renewenvironment{itemize}%
  {\LLlist{1.5em}{0em}{0.25em}{0pt}{0.25em}}{\endlist}
\newenvironment{itemizetight}%
  {\LLlist{1.5em}{0em}{0em}{0pt}{0em}}{\endlist}
\newenvironment{itemizetighter}%
  {\LLlist{1.5em}{0em}{0em}{0pt}{0em}}{\endlist}

% The commentary hung under a publication entry: inset from the right margin,
% held close under its entry (small topsep), and adding no extra space of its
% own after -- the enumerate's itemsep supplies the gap to the next entry.
\newenvironment{itemizetightrightpad}{%
  \list{}{%
    \setlength{\leftmargin}{1.5em}%
    \setlength{\rightmargin}{3em}%
    \setlength{\topsep}{0.15em}%
    \setlength{\partopsep}{0pt}%
    \setlength{\itemsep}{0.1em}%
    \setlength{\parskip}{0pt}%
    \setlength{\parsep}{0em}%
  }%
}{\endlist}

% The numbered publication lists: one consistent gap between entries, whether
% or not an entry carries commentary.  "resume" is accepted on the second list
% so the numbering continues across the subsection break.
\setlist[enumerate]{itemsep=0.45em, parsep=0em, topsep=0.4em, partopsep=0pt}

% Hanging indent for a run-on entry: \parshape 2 3.2em \linewidth \listindent
% \dimexpr\linewidth-\listindent\relax sets the first line full width and
% indents every line after it to \listindent.
\newlength{\listindent}
\setlength{\listindent}{4.8em}

% \changemargin{<left>}{<right>} ... \endchangemargin -- indents a block of the
% document; used to nest the collaboration sub-lists under their headings.
\def\changemargin#1#2{\list{}{\rightmargin#2\leftmargin#1}\item[]}
\let\endchangemargin=\endlist


%% ===========================================================================
%% The right-hand date column
%% ===========================================================================
%% Nearly every entry ends with a date set against the right margin.  \when
%% supplies both the \hfill and the fixed-width box that lines those dates up
%% into one column: the box is left-aligned, so "2024" and "2022--24" begin at
%% the same horizontal position.
%%
%% A \makebox is only as wide as it is declared, so an over-long date used to
%% overflow it silently and print out in the right margin.  The column is sized
%% to hold the widest date in regular use, "2021--2026"; \when still measures
%% each date and sets a wider one ("2022--23, 24--25") flush right instead,
%% which costs the shared left edge on that entry but keeps it inside the text
%% block.

%% The column is measured, not guessed.  Every closed range in the CV is written
%% YYYY--YY, so at 11pt EB Garamond with oldstyle figures the widest date is
%% "2007--09" at 37.6pt, and 3.7em is 40.5pt.  Re-measure whenever the text
%% face, the figure style, or the range convention changes -- digit widths
%% differ between all three, and a column a shade too narrow silently sends its
%% widest dates flush right, which tools/check_alignment.py accepts as
%% legitimate.
\newlength{\datecolumn}
\setlength{\datecolumn}{3.7em}

%% The glue ahead of a date is \datesep plus infinite stretch.  The stretch is
%% what pushes the date to the right margin; the fixed \datesep is a floor, so
%% an entry that grows long enough to reach its date reports an overfull box
%% instead of quietly printing its text hard against the year.  Every entry
%% currently clears it, so the floor costs nothing today -- it is there to catch
%% the next entry that does not.  \whenflush is the fix when one turns up: it
%% gives back the empty right-hand part of the date box, about 1.6em for a
%% four-digit year.
\newlength{\datesep}
\setlength{\datesep}{0.75em}

%% \LLdateglue is the glue that carries the date to the right margin.  It reads
%% as one \hfill most of the time, but is written out so that an entry truly too
%% long to hold its date can break: TeX takes the \penalty300, leaves the text
%% on the line it filled, and the \hbox{} -- which survives the break where glue
%% would not -- carries the second \hfill onto the next line, so the date still
%% lands flush right instead of stranded against the left margin.  \unskip drops
%% the source space ahead of the date, so the gap is \datesep however the entry
%% was typed.
%%
%% The penalty must comfortably exceed \finalhyphendemerits (5000 demerits, i.e.
%% a penalty of ~70): at 50, TeX preferred breaking the date onto its own line
%% over ending an entry's second-to-last line in a hyphen, and every hyphenated
%% publication title shed its year.  At 300 (90000 demerits) the break survives
%% only as the escape hatch it is meant to be.
\newcommand{\LLdateglue}{%
  \unskip\nobreak\hspace{\datesep}\hfill\penalty300 \hbox{}\nobreak\hfill
}

\newlength{\LLdatewidth}
\newcommand{\when}[1]{%
  \LLdateglue
  \settowidth{\LLdatewidth}{#1}%
  \ifdim\LLdatewidth>\datecolumn #1\else\makebox[\datecolumn][l]{#1}\fi
}

% \whenflush{<date>} -- the date at its natural width, flush right.  This is
% the form the publication lists use: their entries are multi-line paragraphs,
% every date there is a bare year (so flush right aligns into a column anyway),
% and the empty tail of a \datecolumn box would only push years onto lines of
% their own.  Also the escape hatch for a single-line entry whose text runs too
% close to the margin for the full column.
\newcommand{\whenflush}[1]{\LLdateglue #1}

% \rightnote{<text>} -- a right-aligned italic annotation (an award, a current
% position) hung underneath an entry.  The trailing pad makes it end where a
% four-digit year in the date column ends, rather than at the margin itself.
\newcommand{\rightnote}[1]{%
  \settowidth{\LLdatewidth}{2024}%
  \hfill\emph{#1}\hspace{\dimexpr\datecolumn-\LLdatewidth\relax}%
}


%% ===========================================================================
%% Live publication metrics from INSPIRE-HEP
%% ===========================================================================
%% Citation counts go stale the moment they are typed, so they are not written
%% into the CV by hand.  The inspirehep package keeps them in a generated file
%% of declarations,
%%
%%     \inspiresetcites{2690093}{3}                        citations of a paper
%%     \inspiresetauthorstat{1071846}{citations}{207838}    an author-level figure
%%
%% which the CV body reads back through \inspirepub and \inspirestat.  The
%% fetcher shipped with the package writes that file: it collects the records by
%% scanning this document for \inspirepub, follows the \input files from the top
%% level, and takes the author from the INSPIRE author link in the publications
%% preamble.  build.sh and CI run it before the first pdflatex pass.  Nothing
%% here touches the network, so the CV still compiles offline, and on Overleaf,
%% from whatever LL_InspireData.tex already holds.
%%
%% The package lives in the inspirehep-latex submodule -- published on its own
%% at https://github.com/lawrenceleejr/inspirehep-latex -- so a fresh clone
%% needs "git submodule update --init" before its first build.
%%
%% inspirehep.sty is found through TEXINPUTS, which build.sh points at the
%% submodule.  The bare name is deliberate: it also resolves when the file sits
%% beside the sources, which is how Overleaf has to see it -- Overleaf's Git
%% sync does not fetch submodules, so there the .sty is uploaded to the project
%% root instead.  A path-qualified \usepackage would find neither.
%%
%% Loaded here rather than beside the packages above because it draws hyperref
%% links, and hyperref has to come last in that list.
\usepackage[
  data     = LL_InspireData,   % the generated figures (.tex is added)
  script   = inspirehep-latex/inspirehep-fetch.py,
  mincites = 1,                % a paper below this is left unannotated
  maxage   = 120,              % warn once the figures are a quarter old
  fetch    = off,              % build.sh runs the fetcher; TeX never shells out
]{inspirehep}

% The citation tag is metadata and should read a level below the title it
% annotates: the package sets it small, and this adds the muted grey.
%
% The annotation costs about a dozen characters on every entry, which is enough
% to push an entry that already filled its line onto a second one.  Raising
% mincites above, or shortening the label, is the dial if that happens too often.
\definecolor{LLmuted}{gray}{0.40}
\renewcommand{\inspirecitesformat}[1]{\nobreakspace{\small\color{LLmuted}[#1]}}

% \inspirepub{<record>}{<title>} -- a publication title in bold, linked to its
% INSPIRE-HEP record and followed by that record's live citation count.  The CV
% quotes each published title in its own wording, maths, and capitalisation
% rather than taking INSPIRE's, and the package accepts that as a title=
% option; wrapping it as a second argument keeps the call sites short.
\providecommand{\NewCommandCopy}{\let}   % for LaTeX older than 2021-06
\NewCommandCopy{\inspirepubfromid}{\inspirepub}
\renewcommand{\inspirepub}[2]{\inspirepubfromid[title={#2}]{#1}}

% \pub{<url>}{<title>} -- the same, for an entry with no INSPIRE-HEP record:
% arXiv-only preprints, CDS and CERN notes, bare DOI links.
\newcommand{\pub}[2]{\href{#1}{\textbf{#2}}}

% \inspirestat{<key>} -- an author-level figure: papers, citations or hindex.
% A missing key is reported loudly rather than left silently blank.
%
% The exact counts move daily and the sentence quoting them says "over", so the
% two large figures are rounded down.  The paper count keeps its digits
% unseparated, as it has read until now; drop its sep=none to give it the
% thousands comma the citation count already has.
\makeatletter
\newcommand{\LL@stat@papers}   {\inspirepapers   [round=100, sep=none]{1071846}}
\newcommand{\LL@stat@citations}{\inspirecitations[round=1000]{1071846}}
\newcommand{\LL@stat@hindex}   {\inspirehindex{1071846}}
\newcommand{\inspirestat}[1]{%
  \ifcsname LL@stat@#1\endcsname\csname LL@stat@#1\endcsname\else\textbf{[?~#1]}\fi
}
\makeatother

% Fallbacks: the last hand-checked figures, for a build that has never run the
% fetcher -- an offline checkout, or a fresh Overleaf project -- so the document
% always compiles with plausible numbers.  These are raw counts; the rounding
% and the thousands separator are applied above.  LL_InspireData.tex is read at
% \begin{document} and overrides every one of them.
\inspiredefaultauthorstat{1071846}{papers}{1467}
\inspiredefaultauthorstat{1071846}{citations}{207838}
\inspiredefaultauthorstat{1071846}{hindex}{211}
